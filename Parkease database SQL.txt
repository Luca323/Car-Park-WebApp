USE parkease;
CREATE TABLE logininfo
(
    UserID INTEGER PRIMARY KEY NOT NULL,
    Username VARCHAR(50) NOT NULL,
    Passkey VARCHAR(100) NOT NULL,
    Email Varchar(30) NOT NULL,
    PhoneNum VARCHAR(15) NOT NULL,
    CarNum VARCHAR(10),
    Type ENUM('driver', 'admin') NOT NULL DEFAULT 'driver'
);

CREATE TABLE Carparks (
     CarparkID INT PRIMARY KEY NOT NULL,
     Name VARCHAR(20),
     Size INT
);

CREATE TABLE Spaces (
    SpaceID INT AUTO_INCREMENT PRIMARY KEY,
    CarparkID INT NOT NULL,
    Price INT NOT NULL,
    Occupied BOOLEAN NOT NULL,
    UserID INT DEFAULT NULL,
    Status ENUM('Available', 'Blocked', 'Reserved', 'Occupied') DEFAULT 'Available',
    FOREIGN KEY (CarparkID) REFERENCES Carparks(CarparkID) ON DELETE CASCADE,
    FOREIGN KEY (UserID) REFERENCES logininfo(UserID) ON DELETE SET NULL
);

CREATE TABLE IF NOT EXISTS Requests (
  ReqID INT AUTO_INCREMENT PRIMARY KEY,
  CarParkID INT NOT NULL,
  UserID INT NOT NULL,
  Start DATETIME NOT NULL,
  End DATETIME NOT NULL,
  Cost DECIMAL(10,2),
  Status VARCHAR(20),
  FOREIGN KEY (CarParkID) REFERENCES Carparks(CarparkID) ON DELETE CASCADE,
  FOREIGN KEY (UserID) REFERENCES logininfo(UserID) ON DELETE CASCADE
);

CREATE TABLE Reservations(
    ReservationID INTEGER PRIMARY KEY NOT NULL,
    UserID INTEGER NOT NULL,
    SpaceID INTEGER NOT NULL,
    startDate DATE NOT NULL,
    Arrival TIME NOT NULL,
    Departure TIME NOT NULL,
    FOREIGN KEY (SpaceID) REFERENCES Spaces(SpaceID) ON DELETE CASCADE,
    FOREIGN KEY (UserID) REFERENCES logininfo(UserID) ON DELETE CASCADE
);

DELIMITER $$

CREATE PROCEDURE create_spaces_for_carpark(IN parkId INT, IN numSpaces INT)
BEGIN
    DECLARE i INT DEFAULT 1;
    WHILE i <= numSpaces DO
        INSERT INTO Spaces (CarparkID, Price, Occupied, UserID)
        VALUES (parkId, 0, FALSE, NULL);
        SET i = i + 1;
    END WHILE;
END$$

CREATE PROCEDURE add_additional_spaces(IN parkId INT, IN oldSize INT, IN newSize INT)
BEGIN
    DECLARE i INT DEFAULT oldSize + 1;
    WHILE i <= newSize DO
        INSERT INTO Spaces (CarparkID, Price, Occupied, UserID)
        VALUES (parkId, 0, FALSE, NULL);
        SET i = i + 1;
    END WHILE;
END$$

CREATE TRIGGER trg_auto_create_spaces
AFTER INSERT ON Carparks
FOR EACH ROW
BEGIN
    CALL create_spaces_for_carpark(NEW.CarparkID, NEW.Size);
END$$

CREATE TRIGGER trg_auto_expand_spaces
AFTER UPDATE ON Carparks
FOR EACH ROW
BEGIN
    IF NEW.Size > OLD.Size THEN
        CALL add_additional_spaces(NEW.CarparkID, OLD.Size, NEW.Size);
    END IF;
END$$

CREATE TRIGGER `check_userid_on_occupied_insert`
BEFORE INSERT ON `Spaces`
FOR EACH ROW
BEGIN
    IF NEW.Occupied = 1 AND NEW.UserID IS NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'UserID must be provided when space is occupied';
    END IF;

    IF NEW.Occupied = 0 AND NEW.UserID IS NOT NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'UserID must be NULL when space is not occupied';
    END IF;
END$$

CREATE TRIGGER `check_userid_on_occupied_update`
BEFORE UPDATE ON `Spaces`
FOR EACH ROW
BEGIN
    IF NEW.Occupied = 1 AND NEW.UserID IS NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'UserID must be provided when space is occupied';
    END IF;

    IF NEW.Occupied = 0 AND NEW.UserID IS NOT NULL THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'UserID must be NULL when space is not occupied';
    END IF;
END$$

DELIMITER ;
